4. **Test infrastructure functions**:

   ```bash
   PASS=0; FAIL=0; TOTAL=0

   assert_status() {
       local test_name="$1"
       local expected="$2"
       local actual="$3"
       TOTAL=$((TOTAL + 1))
       if [ "$actual" -eq "$expected" ]; then
           green "  PASS: $test_name (HTTP $actual)"
           PASS=$((PASS + 1))
       else
           red "  FAIL: $test_name (expected HTTP $expected, got HTTP $actual)"
           FAIL=$((FAIL + 1))
       fi
   }

   # Helper to make requests and capture status code + body
   do_request() {
       local method="$1"; shift
       local url="$1"; shift
       # remaining args are extra curl flags
       RESPONSE=$(curl -s -w "\n%{http_code}" -X "$method" "$url" \
           -H "Content-Type: application/json" \
           "$@")
       HTTP_CODE=$(echo "$RESPONSE" | tail -1)
       BODY=$(echo "$RESPONSE" | sed '$d')
   }
   ```
