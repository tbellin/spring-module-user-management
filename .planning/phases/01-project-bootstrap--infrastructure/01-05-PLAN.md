```sql
-- V1__init_schema.sql
-- Initial schema for user management system
-- Compatible with both H2 (MODE=PostgreSQL) and PostgreSQL

-- User table (named app_user to avoid reserved word conflicts)
CREATE TABLE IF NOT EXISTS app_user (
    id              BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    email           VARCHAR(255) NOT NULL UNIQUE,
    username        VARCHAR(100) NOT NULL UNIQUE,
    password_hash   VARCHAR(255) NOT NULL,
    first_name      VARCHAR(100),
    last_name       VARCHAR(100),
    enabled         BOOLEAN NOT NULL DEFAULT FALSE,
    email_verified  BOOLEAN NOT NULL DEFAULT FALSE,
    created_at      TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at      TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- Role table
CREATE TABLE IF NOT EXISTS app_role (
    id   BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name VARCHAR(50) NOT NULL UNIQUE
);

-- User-Role junction table (many-to-many)
CREATE TABLE IF NOT EXISTS user_role (
    user_id BIGINT NOT NULL,
    role_id BIGINT NOT NULL,
    PRIMARY KEY (user_id, role_id),
    CONSTRAINT fk_user_role_user FOREIGN KEY (user_id) REFERENCES app_user(id) ON DELETE CASCADE,
    CONSTRAINT fk_user_role_role FOREIGN KEY (role_id) REFERENCES app_role(id) ON DELETE CASCADE
);

-- Verification token table (for email verification)
CREATE TABLE IF NOT EXISTS verification_token (
    id          BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    token       VARCHAR(255) NOT NULL UNIQUE,
    user_id     BIGINT NOT NULL,
    expiry_date TIMESTAMP NOT NULL,
    used        BOOLEAN NOT NULL DEFAULT FALSE,
    created_at  TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_verification_token_user FOREIGN KEY (user_id) REFERENCES app_user(id) ON DELETE CASCADE
);

-- Password reset token table
CREATE TABLE IF NOT EXISTS password_reset_token (
    id          BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    token       VARCHAR(255) NOT NULL UNIQUE,
    user_id     BIGINT NOT NULL,
    expiry_date TIMESTAMP NOT NULL,
    used        BOOLEAN NOT NULL DEFAULT FALSE,
    created_at  TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_password_reset_token_user FOREIGN KEY (user_id) REFERENCES app_user(id) ON DELETE CASCADE
);

-- Indexes for performance
CREATE INDEX IF NOT EXISTS idx_app_user_email ON app_user(email);
CREATE INDEX IF NOT EXISTS idx_app_user_username ON app_user(username);
CREATE INDEX IF NOT EXISTS idx_verification_token_token ON verification_token(token);
CREATE INDEX IF NOT EXISTS idx_password_reset_token_token ON password_reset_token(token);

-- Seed default roles
INSERT INTO app_role (name) VALUES ('ROLE_USER');
INSERT INTO app_role (name) VALUES ('ROLE_ADMIN');
```

CRITICAL SQL compatibility notes (from RESEARCH.md Pitfall 2):
- Use GENERATED BY DEFAULT AS IDENTITY (not SERIAL - PostgreSQL-only and deprecated)
- Use BOOLEAN (works on both databases)
- Use TIMESTAMP (not TIMESTAMP WITH TIME ZONE unless needed)
- Use VARCHAR (not TEXT for columns with length limits)
- Use CURRENT_TIMESTAMP (standard SQL)
- Named constraints for better error messages
- IF NOT EXISTS for idempotency (though Flyway handles this via checksums)

Tables created:
- app_user: Core user entity
- app_role: Role definitions
- user_role: Many-to-many junction table
- verification_token: Email verification (Phase 4)
- password_reset_token: Password reset (Phase 5)

Creating token tables now even though they're used in later phases because:
1. Better to have complete schema upfront
2. Avoids schema migration coordination issues later
3. Empty tables have negligible cost
